cmake_minimum_required(VERSION 3.25)
project(P15_netd VERSION 1.0.0 LANGUAGES C)
include(CTest)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_COVERAGE "Enable GCC/Clang coverage" OFF)
option(ENABLE_ASAN "Enable AddressSanitizer" OFF)
option(ENABLE_UBSAN "Enable UndefinedBehaviorSanitizer" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(ENABLE_COVERAGE)
  add_compile_options(--coverage -O0 -g)
  add_link_options(--coverage)
endif()
if(ENABLE_ASAN)
  add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address)
endif()
if(ENABLE_UBSAN)
  add_compile_options(-fsanitize=undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=undefined)
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
find_library(SQLITE3_LIBRARY NAMES sqlite3 sqlite3.so.0 REQUIRED)
add_library(netd_core src/config.c src/util.c src/log.c src/db.c src/protocol.c)
target_include_directories(netd_core PUBLIC include)
target_link_libraries(netd_core PUBLIC OpenSSL::Crypto Threads::Threads ${SQLITE3_LIBRARY})
add_executable(netd src/netd.c src/server.c)
target_link_libraries(netd PRIVATE netd_core)
add_executable(netctl src/netctl.c)
target_link_libraries(netctl PRIVATE netd_core)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(NN IN ITEMS 01 02 03 04 05 06 07 08 09 10 11 12)
    add_test(NAME NETD-${NN} COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_netd_${NN}.py)
    set_tests_properties(NETD-${NN} PROPERTIES ENVIRONMENT "NETD_BUILD_DIR=${CMAKE_BINARY_DIR};NETD_SOURCE_DIR=${CMAKE_SOURCE_DIR}" TIMEOUT 30)
  endforeach()
endif()
